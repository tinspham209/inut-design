### Step 1.4.2 — Create `components/scripts/faq-schema.tsx`

**File:** `components/scripts/faq-schema.tsx` (new)  
**Action:** Reusable component accepting a FAQ items array, renders `FAQPage` JSON-LD.

```tsx
interface FAQItem { question: string; answer: string }
interface Props { items: FAQItem[] }

export function FAQSchema({ items }: Props) {
  const schema = {
    "@context": "https://schema.org",
    "@type": "FAQPage",
    mainEntity: items.map((item) => ({
      "@type": "Question",
      name: item.question,
      acceptedAnswer: { "@type": "Answer", text: item.answer },
    })),
  };
  return <script type="application/ld+json"
    dangerouslySetInnerHTML={{ __html: JSON.stringify(schema) }} />;
}
```
